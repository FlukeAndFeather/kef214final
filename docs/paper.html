Methods

  • Creating a function to find the moving averages
moving_average <- function(hurricane_df) {
  # Initialize a tibble to contain the results
  result <- tibble(
    window_start = seq(
      ymd("1988-01-05"),
      ymd("1994-12-27"),
      by = "9 weeks"),
    k_mgl = NA,
    mg_mgl = NA,
        # Fill in the rest of the ions
    ca_mgl = NA,
    no3n_mgl = NA,
    nh4n_mgl = NA
  )

  for (i in 1:nrow(result)) {
    w1 <- result$window_start[i] 
    w2 <- w1 + weeks(9)

    in_window <- hurricane_df$Sample_Date >= w1 & hurricane_df$Sample_Date < w2

    # Use indexing to pull out the ion concentrations that fall inside the window
    k_window <- hurricane_df$K[in_window]
    mg_window <- hurricane_df$Mg[in_window]
    ca_window <- hurricane_df$Ca[in_window]
    no3n_window <- hurricane_df$`NO3-N`[in_window]
    nh4n_window <- hurricane_df$`NH4-N`[in_window]

    result$k_mgl[i] <- mean(k_window, na.rm = TRUE)
    result$mg_mgl[i] <- mean(mg_window, na.rm = TRUE)
    result$ca_mgl[i] <- mean(ca_window, na.rm = TRUE)
    result$no3n_mgl[i] <- mean(no3n_window, na.rm = TRUE)
    result$nh4n_mgl[i] <- mean(nh4n_window, na.rm = TRUE) 
   }
  return(result)
}
